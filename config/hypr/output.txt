general {
    lock_cmd = omarchy-lock-screen                         # lock screen and 1password
    before_sleep_cmd = loginctl lock-session               # lock before suspend.
    after_sleep_cmd = hyprctl dispatch dpms on HDMI-A-1; hyprctl dispatch dpms on DP-2 && brightnessctl -r # to avoid having to press a key twice to turn on the display.
    inhibit_sleep = 3                                      # wait until screen is locked
}

listener {
    timeout = 150                                             # 2.5min: Start screensaver/lock
    on-timeout = pidof hyprlock || omarchy-launch-screensaver
    # Ensure both monitors turn on if activity resumes before the full screen off timeout
    on-resume = hyprctl dispatch dpms on HDMI-A-1; hyprctl dispatch dpms on DP-2 && brightnessctl -r
}

listener {
    timeout = 300                      # 5min: Lock screen when timeout has passed (redundant if omarchy-lock-screen is always run by the 150s timeout)
    on-timeout = loginctl lock-session # You might want to remove this if 'omarchy-lock-screen' is sufficient and always runs at 150s.
}

listener {
    timeout = 330                                            # 5.5min: Screen off when timeout has passed
    # Explicitly turn off each monitor
    on-timeout = hyprctl dispatch dpms off HDMI-A-1; hyprctl dispatch dpms off DP-2
    # Explicitly turn on each monitor and reset brightness
    on-resume = hyprctl dispatch dpms on HDMI-A-1; hyprctl dispatch dpms on DP-2 && brightnessctl -r
}
